    # Ranking questions.

    # "Which customer has highest sales?"

    )

        group_by
        and re.search(
            r"\b(?:lowest|minimum|least)\b",
            question_norm,

            r"\b(?:sales?|revenue|billing|turnover)\b",
            question_norm,
        )
    )

            and (
